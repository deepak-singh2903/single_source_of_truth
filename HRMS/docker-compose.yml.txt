version: "3.9"

services:
  backend:
    build: ./backend
    container_name: hrms-backend
    ports:
      - "8000:8000"
    env_file:
      - ./backend/.env.example
    depends_on:
      - db

  frontend:
    build: ./frontend
    container_name: hrms-frontend
    ports:
      - "3000:80"
    depends_on:
      - backend

  db:
    image: postgres:15
    container_name: hrms-db
    restart: always
    environment:
      POSTGRES_USER: user
      POSTGRES_PASSWORD: password
      POSTGRES_DB: hrms
    ports:
      - "5432:5432"
    volumes:
      - postgres_data:/var/lib/postgresql/data
